% =============================================================================
% Pandoc LaTeX Template for Book
% =============================================================================
% This template integrates with latex-template.sty and provides
% the document structure for PDF output.
% =============================================================================

\documentclass[11pt,a4paper,oneside]{book}

% Include preamble and macros (latex-template loaded there via TEXINPUTS)
$for(header-includes)$
$header-includes$
$endfor$

% Document metadata
\title{$if(title)$$title$$else$Untitled$endif$}
\author{$if(author)$$author$$else$Anonymous$endif$}
\date{$if(date)$$date$$else$\today$endif$}

% Syntax highlighting for fenced code blocks (defines Shaded/Highlighting)
$if(highlighting-macros)$
$highlighting-macros$
$endif$

% Hyperref setup (skipped in latex-template, so load here)
\usepackage[colorlinks=true,linkcolor=blue,urlcolor=blue]{hyperref}

% Pandoc list spacing
\providecommand{\tightlist}{%
  \setlength{\itemsep}{0pt}\setlength{\parskip}{0pt}}

% Set counters BEFORE document for correct x.y.serial numbering (ch{x}/{y}-*.md)
$if(is-preface)$
\setcounter{part}{-1}
$else$
$if(chapter-num)$
\setcounter{part}{$chapter-num$}
$else$
\setcounter{part}{0}
$endif$
$if(section-num)$
\setcounter{chapter}{$section-num$}
\addtocounter{chapter}{-1}
\setcounter{section}{0}
$else$
\setcounter{chapter}{0}
\setcounter{section}{0}
$endif$
$endif$

\begin{document}

% Title page for book mode
$if(titlepage)$
\maketitle
\tableofcontents
\newpage
$endif$

% Main content
$body$

% Flush pending material for styles that override \clearpage (bookmode)
\ifdefined\realclearpage
\realclearpage
\fi

\end{document}
